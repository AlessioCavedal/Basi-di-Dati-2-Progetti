Classe Spedizione

[V.Spedizione.fasce_stessa_nazione_offerta_non_sovrapposte]

// Per ogni coppia (Offerta, Nazioe) le fasce di spedizione devono avere interalli disgiunti

NOT EXISTS s1, s2, o, n, min1, min2, max1, max2 Spedizione(s1) AND Offerta(o) AND Nazione(n) AND off_sped(o, s1) AND naz_sped(n, s1) AND Spedizione(s2) AND Offerta(o) AND Nazione(n) AND off_sped(o, s2) AND naz_sped(n, s2) AND s1 != s2 AND min(s1, min1) AND min(s2, min2) AND max(s1, max1) AND max(s2, max2) AND Intero(t) AND t >= min1 AND t <= max1 AND t >= min2 AND t <= max2

Classe CartaDiCredito

[V.CartaDiCredito.appartiene_a_chi_ha_effettuato_l’acquisto]

// La carta di credito di un acquisto deve appartenere all’utente ch eha effettuato l’acqusito

ALL c, u, a CartaDiCredito(c) AND Utente(u) AND Acquisto(a) AND acq_cart(a, c) AND ut_acq(u, a) → ut_cart(u, c)

[V.CartaDiCredito.usata_in_acquisto_non_scaduta]

// Una carta di credito usata in un acquisto non deve essere scaduta

ALL c, a, i, sc CartaDiCredito(c) AND Acquisto(a) AND acq_cart(a, c) AND istante_acq(a, i) AND scadenza(c, sc) → i < sc

Classe BuonoRegalo

[V.BuonoRegalo.appartiene_a_chi_ha_effettuato_l’acquisto]

// I buoni regalo usati per pagare un acquisto devono appartenere all’utente che ha effettuato l’acquisto

ALL u, b, a buo_acq(b, a) AND ut_acq(u, a) → buo_ut(b, u)

[V.BuonoRegalo.usato_in_acquisto_attivo]

// Un buono regalo usato in un acquisto deve essere attivo

ALL b, a, iv, fv, ia buo_acq(b, a) AND inizio_validità(b, iv) AND fine_validità(b, fv) AND istante_acq(a, ia) → ia >= iv AND ia <= fv

Classe Utente

[V.Utente.no_amicizia_se_stesso]

// Un utente non può essere sia il mittente che il destinatario della stessa richiesta di amicizia

ALL u, ra Utente(u) AND RichiestaAmicizia(ra) AND invia(u, ra) → NOT riceve(u, ra)

[V.Utente.no_due_richieste_i_amicizia_stessa_coppia_stessa_direzione]

// Non possono esserci due richieste di amicizia tra la stessa coppia di utenti nella stessa direzione

NOT EXISTS u1, u2, r1, r2 AND invia(u1, r1) AND riceve(u2, r1) AND r1 != r2 AND NOT AmiciziaRifiutata(r1) AND invia(u1, r2) AND riceve(u2, r2) AND NOT AmiciziaRifiutata(r2)

oppure equivalentemente:

ALL u1, u2, r1 invia(u1, r1) AND riceve(u2, r1) AND NOT AmiciziaRifiutata(r1) → NOT EXISTS r2 != r1 invia(u1, r2) AND riceve(u2, r2) AND NOT AmiciziaRifiutata(r2)

[V.Utente.no_richiesta_se_già_amici]

// Se due utenti sono già amici non può esistere un altra richiesta di amicizia tra di loro. Consideriamo alnche la simmetria

NOT EXISTS u1, u2, r1, r2 invia(u1, r1) AND riceve(u2, r1) AND AmiciziaAccettata(r1) AND ((invia(u1, r2) AND riceve(u2, r2)) OR (invia(u2, r2) AND riceve(u1, r2))) AND NOT AmiciziaRiutata(r2)

Classe Acquisto

costo_totale() : Real >= 0

// Per calcolare il prezzo totale di ogni acquisto dobbiamo ci serve

  • le offerte acquistate tramite off_acq (con la relativa quantità)
  • la nazione dell’acquisto tramite acq_naz
  • le fasce di spedizione di ogni offerte per quella nazione tramite off_sped e naz_sped
  • il prezzo di ogni offerta

pre: nessuna

post:

Sia n:Nazione dell’acquisto tale che acq_naz(this, n)

        

Sia O = { (o, qt) | Offerta(o) AND off_acq(this, o) AND n(this, o, qt) }

// Costruisce l’insieme dlle coppie (offerta, quanità) dell acquisto this

        

Per ogni (o, qt) in O

Spedizione(o, qt) = { s | Spedizione(s) AND off_sped(o, s) AND naz_sped(s, n) AND EXISTS m [ min(s, m) AND qt >= m ] AND (NOT EXISTS mx [ max(s, mx)] OR EXISTS mx [ max(s, mx) AND qt <= mx ]) }

// Per ogni coppia (o, qt) costruisce l’insieme delle fasce di spedizione applicabili

Sia C = { c | EXISTS o, qt, s, p, ps [ (o, qt) appartiene a O AND s appartiene a S(o, qt) AND prezzo(o, p) AND prezzo(s, ps) AND c = (p + ps) * qt ]}

// Costruisce l’insieme dei costi parziali, uno per ogni offerta acquistata

Result = Somma dei valori di c appartententi a C

// Somma di tutti i costi parziali per ottenere il costo complessivo dell acquisto this

Classe Articolo

prmin() : Real >= 0

pre: EXISTS o Offerta(o) AND off_art(o, this)

post:

// P è l’insieme dei prezzi di tutte le offerte relative all’articolo this attive al momento dell’invocazione dell’operazione

Sia P = { p | EXISTS o, i [ Offerta(o) AND off_art(o, this) AND prezzo(o, p) AND inizio(o, i) AND i <= Adesso AND (NOT EXISTS f [fine(o, f)] OR EXISTS f [ (fine(o, f) AND f >= Adesso)] ) ] }

result = min( p | appartiene a P )
