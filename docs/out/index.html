Classe Tariffa

[V.Tariffa.settore_è_dela_sala_dell_evento]

ALL t, set, sal, e Tariffa(t) AND sett_tar(set, t) AND sala_sett(sal, sett) AND ev_tar(e, t) → ev_sala(e, sal)

Classe Prenotazione

[V.Prenotazione.nessun_conflitto]

NOT EXISTS pos, pren1, pren2, e pren1 != pren2 AND pren_post(pren1, pos) AND pren_post(pren2, pos) AND ev_pren(e, pren1) AND ev_pren(e, pren2)

[V.Prenotazione.posti_nella_sala_dell_evento]

ALL pren, pos, set, sal, e pre_post(pren, pos) AND post_sett(pos, set) AND sala_sett(sal, set) AND ev_pren(e, pren) → ev_sala(e_sal)

[V.Prenotazione.no_evento_passato]

NOT EXISTS e, pren, ie, ip ev_pren(e, pren) AND istante(e, ie) AND istante(pren, ip) AND ie < ip
