Classe Valutazione:

[V.Valutazione.solo_video_visualizzati]

// non si può votare un video non visualizzato

ALL u, v, val Utente(u) AND Video(v) AND Valutazione(u, v, val) → [

EXISTS vis Visualizzazione(vis) AND ut_vis(u, vis) AND vid_vis(v, vis) ]

[V.Valutazione.no_valutazione_proprio_video]

// non si può valutare un proprio video

ALL u, v, val Utente(u) AND Video(v) AND Valutazione(u, v, val) → NOT pubblicare(u, v)

Classe Video:

n_visualizzazioni() : Int >= 0

// questa operazione ritorna il numero di visualizzazioni di un video

Pre: Nessuna

Post: Result = # { vis | Visualizzazione(vis) AND vid_vis(this, vis) }

n_valutazioni() : int >= 0

// questa operazione ritorna il numero di valutazioni di un video

        pre:

                nessuna

        post:

                result = # { val | EXISTS u [ Utente(u) AND Valutazione(val) AND valutazione(u, this, val) ] }

valutazioneMedia() : reale >= 0

// questa operazione ritorna la valutazione media di un video

        pre:

                n_valutazioni() > 0

        post:

                Sia V = { val | EXISTS u [ Utente(u) AND Valutazione(val) AND valutazione(u, this, val) ] }

                Sia S la somma dei valori in V, formalmente S = Somma { Valore | val appartiene a V }

                result = S / n_valutazioni

        

n_risposte() : int >= 0

// questa operazione ritorna il numero di risposte a un determinato video

        pre:

                nessuna

        post:

                result = # { risp | Video(risp) AND risposta_a_video(risp, this) }

                // result insieme dei link r : risposta_a_video tali che this è coinvolto in r secondo il ruolo

   originale

[V.Video.no_risposta_se_stesso]

// un video non può essere in risposta a se stesso

ALL v Video(v) → NOT risposta_a_video(v, v)

[V.Video.autore_no_risponde_se_stesso]

// un utente non può pubblicare un video in risposta a un suo stesso video

ALL risp, orig, u Video(risp) AND Video(orig) AND Utente(u) AND pubblicare(u,risp) AND risposta_a_video(risp, orig) → NOT pubblicare(u, orig)

Classe VideoCensurato:

[V.VideoCensurato.limitazioni]

// un video censurato non può essere visionato, votato, commentato o aggiunto a playlist

ALL v VideoCensurato(v) → [

NOT EXISTS vis (Visualizzazione(vis) AND vid_vis(v, vis)) AND

NOT EXISTS u, val (Utente(u) AND Valutazione(u, v, val)) AND

NOT EXISTS vip (VideoInPlaylsit(vip) AND ele_vid(vip, v))

]

ricercaVideo(cat: Categoria, tags : Tag[1..*], val: 0..5) : Video[0..*]

// permette all’utente di: TuTubi deve consentire la ricerca di video: in particolare, data una categoria,

un insieme di tag, ed un intero v tra 0 e 5, si vogliono restituire tutti i video disponibili di quella categoria che posseggono almeno uno tra i tag indicati, e che abbiano una valutazione media di almeno v (se un video non ha ancora alcuna valutazione, deve essere restituito comunque).

        pre:

                nessuna

        post:

                result = { v | Video(v) AND NOT VideoCensurato(v) AND

Categoria(cat) AND cat_vid(cat, v) AND

EXISTS t [ Tag(t) AND tag_vid(t, v) AND t appartiene a tags ] AND

[ valutazioneMedia(v) >= val  OR n_valutazioni(v) = 0 ] }

                

ricerca_discussione(cat : Categoria) : Video[0..*]

// TuTubi deve poi permettere di cercare, data una categoria, i video di quella categoria che hanno il numero maggiore di video in risposta, al fine di isolare le discussioni più animate tra gli utenti.

        

        pre:

                nessuna

        post:

                result = { v | Video(v) AND NOT VideoCensurato(v) AND

Categoria(cat) AND cat_vid(cat, v) AND

NOT EXISTS v2 [ Video(v2) AND NOT VideoCensurato(v2) AND Categoria(cat) AND cat_vid(cat, v2) AND n_risposte(v2) > n_risposte(v) ] }

aggiungi_video(v:Video, p: Playlist) : Void

// un utente può aggiungere un video ad una sua playlist

        pre:

                Sia u:Utente l’oggetto associato all’istanza di attore che ha invocato l’operazione

                

                Video(v) AND Playlist(p) AND Utente(u) AND Crea(u, p) AND NOT VideoCensurato(v)  

AND NOT EXISTS vip ( VideoInPlaylist(vip) AND contenere(p, vip) AND ele_vid(vip, v) )

                

        post:

                EXISTS Vip [ VideoInPlaylist(vip) AND ele_vid(vip, v) AND contenere(p, vip)

AND vip.oridine = MAX (ordine di tutti i VideoInPlaylist + 1 ) ]
