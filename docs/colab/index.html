Classe TipologiaAbbonamento

[V.TipologiaAbbonamento.sottoscrizione_valida]

// Un abbonamento di una certa tipologia può essere sottoscritto solo nel periodo di vendita della tipologia

ALL a, ta, ia, iv, fv Abbonamento(a) AND TipologiaAbbonamento(ta) AND abb_tip(a, ta) AND inizio(a, ia) AND inizio_vendita(ta, iv) AND fine_vendita(ta, fv) → ia >= iv AND ia <= fv

Classe TipAziendale

[V.TipAziendale.sottoscritto_solo_da_impresa]

// Un abbonamento aziedale può essere sottoscritto solo da un impresa

ALL a, ta, c Abbonamento(a) AND TipAziendale(ta) AND abb_tip(a, ta) AND stipula(c, a) → Impresa(c)

[V.TipAziendale.no_piu_utenti_del_limite]

// Il numero di persone che utilizza un abbonamento aziendale non deve essere maggiore del limite

ALL ta, a, n TipAziendale(ta) AND Abbonamento(a) AND abb_tip(a, ta) AND n_max_utetenti(ta, n) → n_utilizzatori(a) <= n

[V.TipAziendale.utilizzo_corretto]

// Le persone fisiche che utilizzano un abbonamento aziendale devono far parte dell’impresa che lo ha stipulato

ALL p, a, i PersonaFisica(p) AND Abbonamento(a) AND utilizza(p, a) AND Impresa(i) AND stipula(i, a) → pers_imp(p, i)

Classe TipPersonale

[V.TipPersonale.sottoscritto_solo_da_personafisica]

// Un abbonamento aziedale può essere sottoscritto solo da una persona fisica

ALL a, tp, c Abbonamento(a) AND TipPersonale(tp) AND abb_tip(a, tp) AND stipula(c, a) → PersonaFisica(c)

[V.TipPersonale.utilizzo_corretto]

// Un abbonamento personale può essere utilizzato solo dalla persona fisica che lo ha stipulato 

ALL tp, a, p TipPersonale(tp) AND Abbonamento(a) AND abb_tip(a, tp) AND PersonaFisica(p) AND Utilizza(p, a) → stipula(p, a)

[V.TipPersonale.esattamente_un_utente]

// Un Abbonamento personale è utilizzato da esattamente un utente

ALL a, tp Abbonamento(a) AND TipPersonale(tp) AND abb_tip(a, tp) → n_utilizzatori(a, 1)

Classe Abbonamento

n_utilizzatori() : Int >= 0

pre: nessuna

post:

        result = # { p | PersonaFisica(p) AND utilizza(p, this) }

Classe PostazioneLavoro

[V.PostazioneLavoro.assegnata_solo_se_abbonamento_attivo]

// Ogni postazione può essere assegnata as una persona solo se questa ha un abbonamento attivo

ALL post, pers PostazioneLavoro(post) AND PersonaFisica(pers) AND pers_post(pers, post) → EXISTS a, sc [ Abbonamento(a) AND utilizza(pers, a) AND scadenza(a, sc) AND sc >= Adesso ]

Classe Accesso

[V.Accesso.abbonamento_attivo]

// Chi effettua un accesso deve avere un abbonamento attivo

ALL a, p, e acc_pers(a, p) AND entrata(a, e) → EXISTS abb, sc, i [ Abbonameto(abb) AND utilizza(p, a) AND inizio(a, i) AND scadenza(a, sc) AND e <= sc         AND e >= i ]

Classe Servizio

[V.Servizio.utilizzo_solo_se_incluso_nell_abbonamento]

// Un servizio può essere utilizzato da una persona solo se questa ha un abbonamento che lo include

ALL ut, p, s, iu Utilizzo(ut) AND Servizio(s) AND pers_ut(p, ut) AND ut_serv(ut, s) AND inzio(ut, iu) → EXISTS a, ta, sc, i  [ Abbonamento(a) AND TipologiaAbbonamentp(ta) AND abb_tip(a, ta) AND abbserv(ta, s) AND utilizza(p, a) AND scadenza(a, sc) >= iu AND inizio(a, i) AND i <= iu ]
