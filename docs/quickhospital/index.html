Classe Stanza

posti_disponbili(t: DataOra) : Int >= 0

// questa operazione ritorna il numero di posti disponibili in una stanza in un determinato istante di tempo

pre:

        nessuna

post:         

        posti_stanza = # { l | Letto(l) AND let_sta(l, this) }

        

        posti_occupati = # { l | Letto(l) AND Exists r, da [ Ricovero(r) AND ric_let(r, l) AND let_sta(l, this)

AND NOT RicoveroTerminato(r) AND data_accettazione(r, da) AND da <= t ] }

return = posti_stanza - posti_occupati

Classe Letto

[V.Letto.no_stesso_letto]

// impone che in un letto non possano essere assegnati due ricoveri attivi contemporaneamente

ALL l, r1, r2 Letto(l) AND Ricovero(r1) AND Ricovero(r2) AND NOT RicoveroTerminato(r1) AND NOT RicoveroTerminato(r2) AND ric_let(r1, l) AND ric_let(r2, l) → r1 = r2

Classe Ricovero

[V.Ricovero.dimissione_dopo_accettazione]

Specifica Use-Case Itinerario Visite

ItinerarioVisite(m:Medico) : (Intero >=0, Stanza,, Intero >= 0, Intero >= 0) : [0..*]

// Il tipo di ritorno è un insieme di quadruple dove ogni elemento contiene: Posizione dell’ordinamento, Stanza, Settore, Piano

pre: nessuna

post:

Stanze_da_visitare = { ( stanza, piano, settore ) | Exists l, r [ Stanza(stanza) AND Letto(r) AND Ricovero(r)

AND ric_let(r, l) AND AND let_sta(l, s) AND NOT RicoveroTerminato AND med_ric(m, r) AND settore(stanza, settore) AND piano(stanza, piano) ] }

// abbiamo creato un insieme di triple, una tripla entra in Stanze_da_visitare se e solo se esiste un letto e un ricovero tali che:

  • stanza è una stanza che contiene il letto l
  • r è un ricovero attivo su quel letto
  • il ricovero r è in cura dal medico m
  • piano è il piano di stanza e settore è il settore di stanza

// in sostanza è l’insieme delle stanze che un medico deve visitare, non ancora ordinate.

Result = Sorted(Stanze_da_visitare, stanza <=)

// Restituisce l’insieme Stanze_da_visitare oridinato secondo stanza<= aggiungendo la posizione ad ogni suo elemento, di conseguenza il suo tipo di ritorno sarà una quadrupla (posizione, stanza, piano, settore)

// Sorted è una funzione di ordine superiore che non ha bisogno di essere definita

// Argomento di sinistra: insieme da ordinare, argomento di destra: il criterio di ordinamento

stanza<=(stanza1, piano1, settore1, stanza2, piano2, settore2) : Booleano

// Creiamo l’operazione ausiliaria che utilizzeremo come criterio di confronto da utilizzare in Sorted.

// Sorted non sa da sola come confrontare due triple e decidere quale delle due viene prima

// Criterio di ordinamento, definito tramite la seguente funzione, la quale prende in input due triple (Stanza, #piano, #settore) e restituisce true se la prima tripla deve precedere la seconda nell’ordinamento.

pre: nessuna

post:

        

Se piano1 < piano2 OR ( piano1 = piano2 AND settore1 <= settore2 )

allora result = True

        Altrimenti

result = False

// La stanza1 viene prima della stanza2 se ha il piano più basso OPPURE se hanno lo stesso piano e il settore1 è più basso o uguale al settore2

registrazione_ricovero(paz: Paziente, l: Letto, d: Data): Ricovero

pre:

        NOT EXISTS r [ Ricovero(r) AND NOT RicoveroTerminato(r) AND ric_let(r, l) ]

        

post:

        Il dominio di interpretazione viene esteso con l’elemento ALPHA

        

        le relazioni che interpretano i simboli di predicato si modificano come segue

        

        Ricovero(ALPHA)

        paz_ric(paz, ALPHA)

        data_accettazione(ALPHA, d)

        ric_let(ALPHA, l)

        result = ALPHA

dimissioni_ricovero(r: Ricovero, d:Data): Void

        pre:         

                Sia da tale che data_accettazione(r, da)

                NOT RicoveroTerminato(r) AND da <= d

        

        post:

                Le relazioni che interpretano i simboli di predicato si modificano come segue:

                

        RicoveroTerminato(r)

        data_dimissione(r, d)

medici_idonei(s: Specializzazione) : Medico[0..*]

// Data una prestazione richiesta da un paziente esterno (per una specializzazione s), il sistema deve restituire l’insieme dei medici maggiormente idonei a soddisfarla. Il criterio di idoneità è il seguente: se esistono medici con specializzazione primaria pari ad s, il risultato è l’insieme di tali medici. Altrimenti, il risultato è l’insieme dei medici che hanno s tra le loro specializzazioni secondarie.

pre:

nessuna

post:

spec_primaria = { m | Medico(m) AND Specializzazione(s) AND spec_prim(m, s) }

spec_secondaria = { m | Medico(m) AND Specializzazione(s) AND spec(m, s) }

Se spec_primaria non è un insieme vuoto

        allora result = spec_primaria

Altrimenti

        result = spec_secondaria
